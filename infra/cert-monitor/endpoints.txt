# iPredict TLS endpoint inventory — read by infra/cert-monitor/check-certs.py
#
# One certificate-bearing endpoint per line, six '|'-separated fields:
#
#   name | target | tls_source | renewal | owner | mode
#
#   name        unique, lowercase, used in alerts and in the state file
#   target      host:port           -> TLS handshake (port defaults to 443)
#               file:/path/*.pem    -> certificate file(s) on disk (glob allowed)
#                                     (covers internal/MTLS certs that are not
#                                     reachable over the network from here)
#   tls_source  who issues it: vercel | letsencrypt | acme-internal | private-ca | manual
#   renewal     automatic | manual  (automatic still gets monitored — automation
#                                   fails silently far more often than certs do)
#   owner       team/human to ping when an alert fires
#   mode        active = checked on every run
#               pending = known endpoint not yet deployed; printed on every run
#                         so it cannot be forgotten, never alerts
#
# Sections decide the scope: [public] endpoints face the internet, [internal]
# services only speak TLS inside the network (or with a private CA).
#
# Adding a service: add a line here in the same PR that deploys the service.
# An endpoint missing from this file is an endpoint nobody is watching.

[public]
# Vercel fronts the site and all /api/* routes on one host. TLS is issued and
# renewed by Vercel; we still monitor it because a Vercel/domain/DNS change can
# silently take it to zero days.
frontend | ipredict-stellar.vercel.app:443 | vercel | automatic | platform | active

# Custom production domain (apex + www + api). Deployed on Vercel with the
# apex as the primary domain. Flip to active once DNS cutover is complete.
ipredict-apex | ipredict.xyz:443 | vercel | automatic | platform | pending
ipredict-www | www.ipredict.xyz:443 | vercel | automatic | platform | pending

# Standalone API host (oracle submissions, indexer, leaderboard read replica),
# if/when it is split off the Next.js app. Let's Encrypt via certbot.
api | api.ipredict.xyz:443 | letsencrypt | automatic | platform | pending

[internal]
# Internal-only TLS services. None are deployed yet — each line is a required
# slot: when the service ships, change `pending` to `active` in the same PR.
# host:port targets are checked with a normal TLS handshake from the monitor.

# Oracle submission API (Bearer-token endpoint, private network only).
oracle-api | oracle-api.internal:8443 | acme-internal | automatic | oracle | pending

# Indexer / ops admin interface.
indexer-admin | indexer.internal:9443 | private-ca | manual | data | pending

# Container-to-container mTLS (service mesh sidecar / internal load balancer).
mesh | ipredict-mesh.internal:443 | private-ca | manual | platform | pending

# Certificate files on disk: internal CA chain, mounted client certs, and any
# service that terminates TLS from a file rather than from the network.
# The glob matches every PEM bundle installed under /etc/ipredict/certs/.
internal-cert-files | file:/etc/ipredict/certs/*.crt | private-ca | manual | platform | pending
